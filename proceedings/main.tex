% Paper metadata is passed as arguments so each subfile remains self-contained.
% #1 title, #2 authors, #3 institution/programme, #4 email (optional),
% #5 abstract, #6 keywords. An empty title or author list is a hard error.
\newcommand{\PaperHeading}[6]{%
  \ifstrempty{#1}{\PackageError{aai-proceedings}{Missing paper title}{Set the first argument of \string\PaperHeading.}}{}%
  \ifstrempty{#2}{\PackageError{aai-proceedings}{Missing paper authors}{Set the second argument of \string\PaperHeading.}}{}%
  \stepcounter{paper}%
  \setcounter{section}{0}%
  \setcounter{figure}{0}%
  \setcounter{table}{0}%
  \setcounter{equation}{0}%
  \twocolumn[{%
    \begin{center}
      {\LARGE\bfseries\color{AAIblue}#1\par}\vspace{5pt}
      {\normalsize #2\par}\vspace{2pt}
      {\small #3\par}
      \ifstrempty{#4}{}{\small\href{mailto:#4}{#4}\par}
    \end{center}
    \vspace{4pt}
    \begin{minipage}{0.96\textwidth}\small
      \textbf{Abstract.} #5\par\vspace{4pt}
      \textbf{Keywords:} #6
    \end{minipage}\vspace{13pt}
  }]%
  \phantomsection
  \addcontentsline{toc}{section}{#1}%
}
